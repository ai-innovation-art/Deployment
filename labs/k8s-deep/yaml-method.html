    YAML: commands cheat sheet

    The 5 steps

    # 1. Find the kind, apiVersion, and whether it's namespaced
kubectl api-resources | grep -i deployment

# 2. Generate a template (then delete creationTimestamp, {} and status)
kubectl create deployment menu-api --image=nginx --replicas=3 --dry-run=client -o yaml > my.yaml

# 3. Find where a field goes, and its type
kubectl explain deployment.spec.template.spec.containers
kubectl explain deployment --recursive | less

# 4. Validate against the real API server (nothing is created)
kubectl apply -f my.yaml --dry-run=server

# 5. Compare with what's running, then apply
kubectl diff -f my.yaml
kubectl apply -f my.yaml

    Generators: Namespace, ConfigMap, Secret

    kubectl create namespace quickbite --dry-run=client -o yaml > ns.yaml

kubectl create configmap quickbite-config -n quickbite \
  --from-literal=DB_HOST=postgres --from-literal=CACHE_TTL_SECONDS=30 \
  --dry-run=client -o yaml > cm.yaml

kubectl create secret generic postgres-credentials -n quickbite \
  --from-literal=username=quickbite --from-literal=password=quickbite-dev-password \
  --dry-run=client -o yaml > secret.yaml

    Remember

    ConfigMap values are plain text. A Secret's stringData is plain text, and its data must be base64 (echo -n 'pw' | base64). base64 is encoding, not encryption.
